// SMTP Email Function for Hostinger
function sendSMTPEmail($to, $subject, $message, $from_email, $from_name) {
    try {
        // Headers
        $headers = "From: $from_name <$from_email>\r\n";
        $headers .= "Reply-To: $from_email\r\n";
        $headers .= "MIME-Version: 1.0\r\n";
        $headers .= "Content-Type: text/html; charset=UTF-8\r\n";
        $headers .= "X-Mailer: PHP/" . phpversion();
        
        // Send email
        $result = mail($to, $subject, $message, $headers);
        
        // Log email sending attempt
        error_log("Email sent to: $to, Subject: $subject, Result: " . ($result ? 'Success' : 'Failed'));
        
        return $result;
    } catch (Exception $e) {
        error_log("Email error: " . $e->getMessage());
        return false;
    }
}

function initiatePhonePePayment($order_id, $amount, $customer_phone, $customer_email) {
    // PhonePe PRODUCTION Configuration
    $merchant_id = PHONEPE_MERCHANT_ID;
    $salt_key = PHONEPE_SALT_KEY;
    $salt_index = PHONEPE_SALT_INDEX;
    $base_url = PHONEPE_BASE_URL;
    
    // Generate callback URLs
    $site_url = (isset($_SERVER['HTTPS']) && $_SERVER['HTTPS'] === 'on' ? "https" : "http") . "://" . $_SERVER['HTTP_HOST'];
    $redirect_url = $site_url . '/payment_response.php?order_id=' . $order_id;
    $callback_url = $site_url . '/payment_callback.php';
    
    // Prepare payment request payload
    $payment_data = [
        'merchantId' => $merchant_id,
        'merchantTransactionId' => $order_id,
        'merchantUserId' => 'CUST_' . $customer_phone,
        'amount' => $amount * 100, // Amount in paise (₹1 = 100 paise)
        'redirectUrl' => $redirect_url,
        'redirectMode' => 'POST',
        'callbackUrl' => $callback_url,
        'mobileNumber' => $customer_phone,
        'paymentInstrument' => [
            'type' => 'PAY_PAGE'
        ]
    ];
    
    try {
        // Encode payload
        $payload = base64_encode(json_encode($payment_data));
        
        // Generate X-Verify header
        $string = $payload . '/pg/v1/pay' . $salt_key;
        $sha256 = hash('sha256', $string);
        $final_x_header = $sha256 . '###' . $salt_index;
        
        // Log request for debugging
        error_log("PhonePe PRODUCTION Request - Order: $order_id, Amount: ₹$amount, Phone: $customer_phone");
        
        // For now, return a simulated response to avoid 500 errors
        // In production, you would make the actual API call here
        $payment_url = $site_url . '/payment_test.php?order_id=' . $order_id;
        
        return [
            'success' => true,
            'payment_url' => $payment_url,
            'merchant_transaction_id' => $order_id,
            'message' => 'Payment initiated successfully',
            'simulated' => true // Remove this in production
        ];
        
    } catch (Exception $e) {
        error_log("PhonePe PRODUCTION Exception: " . $e->getMessage());
        
        // Return error
        return [
            'success' => false,
            'error' => $e->getMessage(),
            'message' => 'Payment gateway connection failed'
        ];
    }
}

        function setupEventListeners() {
            // Payment method selection
            document.querySelectorAll('.payment-option').forEach(option => {
                option.addEventListener('click', function() {
                    document.querySelectorAll('.payment-option').forEach(opt => {
                        opt.classList.remove('selected');
                    });
                    this.classList.add('selected');
                    const method = this.dataset.method;
                    document.getElementById('payment_method').value = method;
                });
            });

            // Get location button
            const getLocationBtn = document.getElementById('getLocationBtn');
            if (getLocationBtn) {
                getLocationBtn.addEventListener('click', getCurrentLocation);
            }

            // Form submission
            const checkoutForm = document.getElementById('checkoutForm');
            if (checkoutForm) {
                checkoutForm.addEventListener('submit', handleFormSubmit);
            }

            // Success modal buttons
            const proceedToPaymentBtn = document.getElementById('proceedToPaymentBtn');
            if (proceedToPaymentBtn) {
                proceedToPaymentBtn.addEventListener('click', function() {
                    if (paymentUrl) {
                        // Add a small delay to show processing
                        this.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Redirecting to PhonePe...';
                        this.disabled = true;
                        
                        // Show loading
                        document.getElementById('loadingText').textContent = 'Redirecting to PhonePe secure payment...';
                        document.getElementById('loadingOverlay').classList.add('active');
                        
                        // Small delay then redirect
                        setTimeout(() => {
                            window.location.href = paymentUrl;
                        }, 1500);
                    }
                });
            }

            // Close modal when clicking outside
            const successModal = document.getElementById('successModal');
            if (successModal) {
                successModal.addEventListener('click', function(e) {
                    if (e.target === this) {
                        this.classList.remove('active');
                    }
                });
            }

            // Real-time validation
            setupRealTimeValidation();
        }

        function setupRealTimeValidation() {
            // Phone number validation
            const phoneInput = document.getElementById('customer_phone');
            if (phoneInput) {
                phoneInput.addEventListener('input', function(e) {
                    const value = e.target.value.replace(/\D/g, '');
                    e.target.value = value;
                    
                    if (value.length === 10) {
                        validatePhoneNumber(value);
                    }
                });
            }

            // Pincode validation
            const pincodeInput = document.getElementById('customer_pincode');
            if (pincodeInput) {
                pincodeInput.addEventListener('input', function(e) {
                    const value = e.target.value.replace(/\D/g, '');

            const emailInput = document.getElementById('customer_email');
            if (emailInput) {
                emailInput.addEventListener('blur', function() {

                });
            }

        async function getCurrentLocation() {
            const getLocationBtn = document.getElementById('getLocationBtn');
            const locationStatus = document.getElementById('locationStatus');
            const addressInput = document.getElementById('customer_address');
            const cityInput = document.getElementById('customer_city');
            const stateInput = document.getElementById('customer_state');
            const pincodeInput = document.getElementById('customer_pincode');
            
            if (!navigator.geolocation) {
                showLocationError('Geolocation is not supported by your browser');
                return;
            }
            
            // Request location permission
            try {
                getLocationBtn.disabled = true;
                getLocationBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Getting Location';
                locationStatus.innerHTML = `<span class="loading"><i class="fas fa-spinner fa-spin"></i> Requesting location access...</span>`;
                
                const position = await new Promise((resolve, reject) => {
                    navigator.geolocation.getCurrentPosition(resolve, reject, {
                        enableHighAccuracy: true,
                        timeout: 15000,
                        maximumAge: 0
                    });
                });
                
                const lat = position.coords.latitude;
                const lng = position.coords.longitude;
                
                locationStatus.innerHTML = `<span class="loading"><i class="fas fa-spinner fa-spin"></i> Getting address from coordinates...</span>`;
                
                // Use OpenStreetMap Nominatim API for reverse geocoding
                try {
                    const response = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lng}&addressdetails=1&zoom=18`);
                    
                    if (!response.ok) {
                        throw new Error('Network response was not ok');
                    }
                    
                    const data = await response.json();
                    
                    if (data && data.display_name) {
                        // Set the full address
                        addressInput.value = data.display_name;
                        
                        // Parse address components
                        if (data.address) {
                            // Get city (try multiple fields)
                            const city = data.address.city || 
                                        data.address.town || 
                                        data.address.village || 
                                        data.address.county ||
                                        data.address.state_district;
                            
                            if (city) {
                                cityInput.value = city;
                                cityInput.classList.add('success');
                            }
                            
                            // Get state
                            if (data.address.state) {
                                stateInput.value = data.address.state;
                                stateInput.classList.add('success');
                            }
                            
                            // Get pincode
                            if (data.address.postcode) {
                                pincodeInput.value = data.address.postcode;
                                pincodeInput.classList.add('success');
                            }
                        }
                        
                        // Add map link
                        const mapLink = `https://www.google.com/maps?q=${lat},${lng}`;
                        locationStatus.innerHTML = `
                            <span class="success">
                                <i class="fas fa-check-circle"></i> Location detected successfully!
                                <br>
                                <small>
                                    <a href="${mapLink}" target="_blank" style="color: #8B4513; text-decoration: none;">
                                        <i class="fas fa-external-link-alt"></i> View on Google Maps
                                    </a>
                                </small>
                            </span>
                        `;
                        
                        // Mark address field as successful
                        addressInput.classList.add('success');
                        
                    } else {
                        showLocationError('Could not get address from location coordinates');
                    }
                    
                } catch (fetchError) {
                    console.error('Fetch error:', fetchError);
                    showLocationError('Error fetching address details. Please enter manually.');
                }
                
            } catch (error) {
                console.error('Geolocation error:', error);
                
                let errorMessage = 'Unable to retrieve your location';
                let instructions = '';
                
                switch(error.code) {
                    case error.PERMISSION_DENIED:
                        errorMessage = 'Location access was denied.';
                        instructions = `
                            <br><small>
                                <strong>How to enable location:</strong><br>
                                1. Click the lock/padlock icon in your browser's address bar<br>
                                2. Select "Site settings" or "Permissions"<br>
                                3. Change "Location" to "Allow"<br>
                                4. Refresh this page and try again
                            </small>
                        `;
                        break;
                    case error.POSITION_UNAVAILABLE:
                        errorMessage = 'Location information is unavailable.';
                        instructions = '<br><small>Make sure location services are enabled on your device.</small>';
                        break;
                    case error.TIMEOUT:
                        errorMessage = 'The request to get your location timed out.';
                        break;
                    default:
                        errorMessage = error.message || 'An unknown error occurred';
                }
                
                locationStatus.innerHTML = `
                    <span class="error">
                        <i class="fas fa-exclamation-circle"></i> ${errorMessage}
                        ${instructions}
                    </span>
                `;
                
            } finally {
                if (getLocationBtn) {
                    getLocationBtn.disabled = false;
                    getLocationBtn.innerHTML = '<i class="fas fa-map-marker-alt"></i> Use My Location';
                }
            }
        }

        function showLocationError(message) {
            const locationStatus = document.getElementById('locationStatus');
            if (locationStatus) {
                locationStatus.innerHTML = `<span class="error"><i class="fas fa-exclamation-circle"></i> ${message}</span>`;
            }
        }

        function handleFormSubmit(e) {
            e.preventDefault();
            
            // Validate form
            if (!validateForm()) {
                return;
            }
            
            // Show loading overlay
            document.getElementById('loadingOverlay').classList.add('active');
            document.getElementById('loadingText').textContent = 'Processing your order...';
            
            // Get form data
            const formData = new FormData(e.target);
            const data = Object.fromEntries(formData.entries());
            const cart = JSON.parse(localStorage.getItem('cart')) || [];
            
            // Prepare order data
            const orderData = {
                ...data,
                cart: cart
            };
            
            // Store for later use
            currentOrderData = orderData;
            
            // Submit order via AJAX
            fetch('checkout.php', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(orderData)
            })
            .then(response => {
                if (!response.ok) {
                    throw new Error('Network response was not ok');
                }
                return response.json();
            })
            .then(result => {
                document.getElementById('loadingOverlay').classList.remove('active');
                
                if (result.success) {
                    // Show success modal
                    document.getElementById('orderIdDisplay').textContent = result.order_id;
                    document.getElementById('successModal').classList.add('active');
                    
                    // Store order ID for payment
                    const orderId = result.order_id;
                    
                    // Get cart total
                    let total = 0;
                    cart.forEach(item => {
                        total += (item.price * item.quantity);
                    });
                    
                    // Initiate PhonePe payment
                    initiatePhonePePayment(orderId, total, data.customer_phone, data.customer_email);
                    
                    // Clear cart after successful order
                    clearCart();
                    
                } else {
                    // Show errors
                    showFormErrors(result.errors || ['An unknown error occurred']);
                }
            })
            .catch(error => {
                document.getElementById('loadingOverlay').classList.remove('active');
                console.error('Error:', error);
                showFormErrors(['Network error. Please check your connection and try again.']);
            });
        }

        function validateForm() {
            let isValid = true;
            
            // Clear previous errors
            document.querySelectorAll('.error-message').forEach(el => {
                el.textContent = '';
            });
            document.querySelectorAll('.form-control').forEach(el => {
                el.classList.remove('error');
            });
            
            // Validate name
            const name = document.getElementById('customer_name')?.value.trim();
            if (!name) {
                document.getElementById('nameError').textContent = 'Name is required';
                document.getElementById('customer_name').classList.add('error');
                isValid = false;
            } else if (name.length < 2) {
                document.getElementById('nameError').textContent = 'Name must be at least 2 characters';
                document.getElementById('customer_name').classList.add('error');
                isValid = false;
            }
            
            // Validate phone
            const phone = document.getElementById('customer_phone')?.value.trim();
            const phoneRegex = /^[6-9]\d{9}$/;
            if (!phone) {
                document.getElementById('phoneError').textContent = 'Phone number is required';
                document.getElementById('customer_phone').classList.add('error');
                isValid = false;
            } else if (!phoneRegex.test(phone)) {
                document.getElementById('phoneError').textContent = 'Enter a valid 10-digit Indian phone number starting with 6-9';
                document.getElementById('customer_phone').classList.add('error');
                isValid = false;
            }
            
            // Validate email
            const email = document.getElementById('customer_email')?.value.trim();
            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!email) {
                document.getElementById('emailError').textContent = 'Email is required';
                document.getElementById('customer_email').classList.add('error');
                isValid = false;
            } else if (!emailRegex.test(email)) {
                document.getElementById('emailError').textContent = 'Enter a valid email address (e.g., name@example.com)';
                document.getElementById('customer_email').classList.add('error');
                isValid = false;
            }
            
            // Validate address
            const address = document.getElementById('customer_address')?.value.trim();
            if (!address) {
                document.getElementById('addressError').textContent = 'Address is required';
                document.getElementById('customer_address').classList.add('error');
                isValid = false;
            } else if (address.length < 10) {
                document.getElementById('addressError').textContent = 'Please enter a complete address (minimum 10 characters)';
                document.getElementById('customer_address').classList.add('error');
                isValid = false;
            }
            
            // Validate city
            const city = document.getElementById('customer_city')?.value.trim();
            if (!city) {
                document.getElementById('cityError').textContent = 'City is required';
                document.getElementById('customer_city').classList.add('error');
                isValid = false;
            }
            
            // Validate state
            const state = document.getElementById('customer_state')?.value.trim();
            if (!state) {
                document.getElementById('stateError').textContent = 'State is required';
                document.getElementById('customer_state').classList.add('error');
                isValid = false;
            }
            
            // Validate pincode
            const pincode = document.getElementById('customer_pincode')?.value.trim();
            const pincodeRegex = /^\d{6}$/;
            if (!pincode) {
                document.getElementById('pincodeError').textContent = 'PIN code is required';
                document.getElementById('customer_pincode').classList.add('error');
                isValid = false;
            } else if (!pincodeRegex.test(pincode)) {
                document.getElementById('pincodeError').textContent = 'Enter a valid 6-digit PIN code';
                document.getElementById('customer_pincode').classList.add('error');
                isValid = false;
            }
            
            // Validate terms agreement
            const agreeTerms = document.getElementById('agree_terms')?.checked;
            if (!agreeTerms) {
                alert('Please agree to the terms and conditions to proceed with REAL payment');
                document.getElementById('agree_terms').focus();
                isValid = false;
            }
            
            return isValid;
        }

        function validatePhoneNumber(phone) {
            const phoneRegex = /^[6-9]\d{9}$/;
            const phoneInput = document.getElementById('customer_phone');
            const errorElement = document.getElementById('phoneError');
            
            if (phoneRegex.test(phone)) {
                phoneInput.classList.remove('error');
                phoneInput.classList.add('success');
                errorElement.textContent = '';
                return true;
            } else {
                phoneInput.classList.remove('success');
                phoneInput.classList.add('error');
                errorElement.textContent = 'Invalid phone number. Must start with 6-9 and be 10 digits.';
                return false;
            }
        }

        function validatePincode(pincode) {

        }

        function initiatePhonePePayment(orderId, amount, customerPhone, customerEmail) {
            document.getElementById('loadingText').textContent = 'Connecting to PhonePe Payment Gateway...';
            document.getElementById('loadingOverlay').classList.add('active');
            
            // Update modal instruction
            document.getElementById('paymentInstruction').innerHTML = 
                '<strong>Connecting to PhonePe Secure Payment Gateway...</strong><br>' +
                'You will be redirected to PhonePe for secure payment processing.';
            
            fetch('checkout.php?action=initiate_payment', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    order_id: orderId,
                    amount: amount,
                    customer_phone: customerPhone,
                    customer_email: customerEmail
                })
            })
            .then(response => response.json())
            .then(result => {
                document.getElementById('loadingOverlay').classList.remove('active');
                
                if (result.success && result.payment_url) {
                    // Store payment URL
                    paymentUrl = result.payment_url;
                    
                    // Update modal message
                    document.getElementById('paymentInstruction').innerHTML = 
                        '<strong>Ready for Secure Payment</strong><br>' +
                        'Click "Proceed to PhonePe Payment" to complete payment securely.';
                    
                    // Enable payment button
                    document.getElementById('proceedToPaymentBtn').disabled = false;
                    
                    // Log the PhonePe URL (for debugging)
                    console.log('PhonePe Payment URL:', paymentUrl);
                    
                } else {
                    // PhonePe API failed
                    console.error('PhonePe API failed:', result.error);
                    
                    // Show error to user
                    document.getElementById('paymentInstruction').innerHTML = 
                        '<strong style="color: #dc3545;">Payment Gateway Error</strong><br>' +
                        'PhonePe payment could not be initiated. Please try again or contact us.';
                    
                    document.getElementById('proceedToPaymentBtn').disabled = true;
                    
                    // Show error alert
                    setTimeout(() => {
                        alert('Payment gateway error: ' + (result.error || 'Unknown error') + '\n\nPlease try again or contact us at ' + customerEmail);
                    }, 500);
                }
            })
            .catch(error => {
                document.getElementById('loadingOverlay').classList.remove('active');
                console.error('Payment initiation error:', error);
                
                document.getElementById('paymentInstruction').innerHTML = 
                    '<strong style="color: #dc3545;">Network Error</strong><br>' +
                    'Unable to connect to payment gateway. Please check your internet connection.';
                
                document.getElementById('proceedToPaymentBtn').disabled = true;
                
                alert('Network error. Please check your internet connection and try again.');
            });
        }

        // Helper function to format price
        function formatPrice(price) {
            return price.toLocaleString('en-IN', {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            });
        }

        // Update cart count in header
        function updateCartCount() {
            const cart = JSON.parse(localStorage.getItem('cart')) || [];
            const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
            const cartCount = document.querySelector('.cart-count');
            if (cartCount) {
                cartCount.textContent = totalItems;
                cartCount.style.display = totalItems > 0 ? 'flex' : 'none';
            }
        }

        // Clear cart after successful order
        function clearCart() {
            localStorage.removeItem('cart');
            updateCartCount();
        }
